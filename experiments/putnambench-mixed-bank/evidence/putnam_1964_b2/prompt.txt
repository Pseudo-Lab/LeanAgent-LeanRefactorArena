/--
**Alexander's subbasis theorem**. Suppose `X` is a topological space with a subbasis `S` and `s` is

    {S : Set (Set X)} (hTS : T = generateFrom S) {s : Set X}
    (h : ∀ P ⊆ S, s ⊆ ⋃₀ P → ∃ Q ⊆ P, Q.Finite ∧ s ⊆ ⋃₀ Q) :
    IsCompact s := by

  intro F hsF
  by_contra hF

  choose! U hxU hSU hUF using hSF
