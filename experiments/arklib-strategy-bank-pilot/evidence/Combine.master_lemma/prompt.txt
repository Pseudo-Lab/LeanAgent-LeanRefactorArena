def degCor
  (φ : ι ↪ F) (dstar degree : ℕ) (r : F) (f : ι → F) (x : ι) : F :=
    f x * ∑ l ∈ range (dstar - degree + 1), ((φ x) * r) ^ l

  let q := φ x * r
  degCor φ dstar degree r f x =
    if q ≠ 1

  · rw [Nat.cast_add, Nat.cast_sub hd, Nat.cast_one]
    split_ifs <;> ring
